const currencyFormatter =

            style: "currency",
            currency: "INR",
            minimumFractionDigits: 2,
            maximumFractionDigits: 2
        }

function formatCurrency(value) {

    return currencyFormatter.format(
        Number(value) || 0
    );

}


/* =====================================================
   DATE HELPERS
===================================================== */

function getTodayString() {

    const now = new Date();

    const year =
        now.getFullYear();

    const month =
        String(
            now.getMonth() + 1
        ).padStart(2, "0");

    const day =
        String(
            now.getDate()
        ).padStart(2, "0");

    return `${year}-${month}-${day}`;

}


function formatDate(dateString) {

    const date =
        new Date(
            `${dateString}T00:00:00`
        );

    if (
        Number.isNaN(
            date.getTime()
        )
    ) {
        return dateString;
    }

    return date.toLocaleDateString(
        "en-IN",
        {
            day: "numeric",
            month: "short",
            year: "numeric"
        }
    );

}


/* =====================================================
   TOAST
===================================================== */

let toastTimer;

function showToast(message) {

    toast.textContent = message;

    toast.classList.add("show");

    clearTimeout(toastTimer);

    toastTimer =
        setTimeout(
            () => {
                toast.classList.remove("show");
            },
            2500
        );

}


/* =====================================================
   LOAD DATA SAFELY
===================================================== */

function loadTransactions() {

    try {

        const saved =
            localStorage.getItem(
                STORAGE_KEY
            );

        if (!saved) {
            return [];
        }

        const parsed =
            JSON.parse(saved);

        if (!Array.isArray(parsed)) {
            return [];
        }

        return parsed.filter(
            item =>
                item &&
                typeof item === "object" &&
                Number.isFinite(
                    Number(item.amount)
                ) &&
                Number(item.amount) > 0 &&
                (
                    item.type === "income" ||
                    item.type === "expense"
                )
        ).map(item => ({

            id:
                Number(item.id) ||
                Date.now() +
                Math.random(),

            description:
                String(
                    item.description || "Transaction"
                ).slice(0, 100),

            amount:
                Number(item.amount),

            type:
                item.type,

            category:
                String(
                    item.category || "Other"
                ),

            date:
                String(
                    item.date ||
                    getTodayString()
                )

        }));

    } catch (error) {

        console.error(
            "Could not load transactions:",
            error
        );

        return [];

    }

}

/* =====================================================
   FORM SUBMIT
===================================================== */

form.addEventListener(
    "submit",
    function(e) {

        e.preventDefault();

        const text =
            description.value.trim();

        const numericAmount =
            Number(amount.value);

        const selectedType =
            type.value;

        const selectedCategory =
            category.value;

        const selectedDate =
            transactionDate.value;

        /* -------------------------
           VALIDATION
        ------------------------- */

        if (!text) {

            showToast(
                "Please enter a description."
            );

            description.focus();

            return;

        }

        if (
            !Number.isFinite(
                numericAmount
            ) ||
            numericAmount <= 0
        ) {

            showToast(
                "Please enter a valid amount."
            );

            amount.focus();

            return;

        }

        if (
            numericAmount >
            1000000000
        ) {

            showToast(
                "Amount is too large."
            );

            amount.focus();

            return;

        }

        if (!selectedDate) {

            showToast(
                "Please select a date."
            );

            transactionDate.focus();

            return;

        }


        /* -------------------------
           EDIT EXISTING
        ------------------------- */

        if (editingId !== null) {

            const index =
                transactions.findIndex(
                    item =>
                        item.id === editingId
                );

            if (index !== -1) {

                transactions[index] = {

                    ...transactions[index],

                    description:
                        text,

                    amount:
                        numericAmount,

                    type:
                        selectedType,

                    category:
                        selectedCategory,

                    date:
                        selectedDate

                };

                showToast(
                    "Transaction updated."
                );

            }

            cancelEdit();

        }

        /* -------------------------
           ADD NEW
        ------------------------- */

        else {

            const newTransaction = {

                id:
                    Date.now(),

                description:
                    text,

                amount:
                    numericAmount,

                type:
                    selectedType,

                category:
                    selectedCategory,

                date:
                    selectedDate

            };

            transactions.push(
                newTransaction
            );

            showToast(
                "Transaction added."
            );

        }


        saveData();

        form.reset();

        transactionDate.value =
            getTodayString();

        type.value =
            "expense";

        displayTransactions();

    }
);


/* =====================================================
   DISPLAY TRANSACTIONS
===================================================== */

function displayTransactions() {

    transactionList.innerHTML = "";

    const selectedFilter =
        filter.value;

    let filtered =
        transactions.slice();

    if (
        selectedFilter !== "all"
    ) {

        filtered =
            filtered.filter(
                item =>
                    item.type ===
                    selectedFilter
            );

    }

    filtered.sort(
        (a, b) => {

            const dateDifference =
                new Date(b.date) -
                new Date(a.date);

            if (
                dateDifference !== 0
            ) {
                return dateDifference;
            }

            return (
                Number(b.id) -
                Number(a.id)
            );

        }
    );


    if (
        filtered.length === 0
    ) {

        emptyMessage.style.display =
            "block";

    } else {

        emptyMessage.style.display =
            "none";

    }


    filtered.forEach(
        item => {

            const div =
                document.createElement(
                    "div"
                );

            div.className =
                "transaction";


            /* -------------------------
               LEFT SIDE
            ------------------------- */

            const left =
                document.createElement(
                    "div"
                );

            left.className =
                "transaction-left";


            const icon =
                document.createElement(
                    "div"
                );

            icon.className =
                "transaction-icon";

            icon.textContent =
                getCategoryIcon(
                    item.category
                );


            const info =
                document.createElement(
                    "div"
                );


            const name =
                document.createElement(
                    "div"
                );

            name.className =
                "transaction-name";

            /*
             * IMPORTANT:
             * textContent is used instead of
             * innerHTML to prevent HTML/JS injection.
             */

            name.textContent =
                item.description;


            const categoryText =
                document.createElement(
                    "div"
                );

            categoryText.className =
                "transaction-category";

            categoryText.textContent =
                `${item.category} • ${formatDate(item.date)}`;


            info.appendChild(name);
            info.appendChild(categoryText);

            left.appendChild(icon);
            left.appendChild(info);


            /* -------------------------
               RIGHT SIDE
            ------------------------- */

            const right =
                document.createElement(
                    "div"
                );

            right.className =
                "transaction-right";


            const amountElement =
                document.createElement(
                    "div"
                );

            amountElement.className =
                "amount " +
                (
                    item.type === "income"
                        ? "income"
                        : "expense"
                );

            amountElement.textContent =
                (
                    item.type === "income"
                        ? "+"
                        : "-"
                ) +
                formatCurrency(
                    item.amount
                );


            const actions =
                document.createElement(
                    "div"
                );

            actions.className =
                "action-buttons";


            const editButton =
                document.createElement(
                    "button"
                );

            editButton.className =
                "edit-btn";

            editButton.type =
                "button";

            editButton.textContent =
                "Edit";

            editButton.addEventListener(
                "click",
                () =>
                    editTransaction(
                        item.id
                    )
            );


            const deleteButton =
                document.createElement(
                    "button"
                );

            deleteButton.className =
                "delete-btn";

            deleteButton.type =
                "button";

            deleteButton.textContent =
                "Delete";

            deleteButton.addEventListener(
                "click",
                () =>
                    deleteTransaction(
                        item.id
                    )
            );


            actions.appendChild(
                editButton
            );

            actions.appendChild(
                deleteButton
            );


            right.appendChild(
                amountElement
            );

            right.appendChild(
                actions
            );


            div.appendChild(left);
            div.appendChild(right);

            transactionList.appendChild(
                div
            );

        }
    );


    updateSummary();

    createGraph();

}

    const icons = {

        Shopping: "🛍️",

        Bills: "📄",

        Education: "📚",

        Entertainment: "🎮",

        Other: "💰"

    };

        "💰"
    );

/* =====================================================
   DELETE
===================================================== */

function deleteTransaction(id) {

    const transaction =
        transactions.find(
            item =>
                item.id === id
        );

    if (!transaction) {
        return;
    }


    const confirmed =
        confirm(
            `Delete "${transaction.description}"?`
        );

    if (!confirmed) {
        return;
    }


    transactions =
        transactions.filter(
            item =>
                item.id !== id
        );


    saveData();

    displayTransactions();

    showToast(
        "Transaction deleted."
    );

}


/* =====================================================
   EDIT
===================================================== */

function editTransaction(id) {

    const transaction =
        transactions.find(
            item =>
                item.id === id
        );

    if (!transaction) {
        return;
    }


    editingId =
        id;


    description.value =
        transaction.description;

    amount.value =
        transaction.amount;

    type.value =
        transaction.type;

    category.value =
        transaction.category;

    transactionDate.value =
        transaction.date;


    formHeading.textContent =
        "Edit Transaction";

    submitBtn.textContent =
        "✓ Update Transaction";


    description.focus();

}


/* =====================================================
   CANCEL EDIT
===================================================== */

function cancelEdit() {

    editingId =
        null;

    formHeading.textContent =
        "Add Transaction";

    submitBtn.textContent =
        "+ Add Transaction";

}


/* =====================================================
   SUMMARY
===================================================== */

function updateSummary() {

    let totalIncome =
        0;

    let totalExpense =
        0;


    transactions.forEach(
        item => {

            /*
             * Convert explicitly to Number
             * before adding.
             */

            const value =
                Number(item.amount);

            if (
                !Number.isFinite(value)
            ) {
                return;
            }


            if (
                item.type === "income"
            ) {

                totalIncome +=
                    value;

            } else if (
                item.type === "expense"
            ) {

                totalExpense +=
                    value;

            }

        }
    );


    income.textContent =
        formatCurrency(
            totalIncome
        );

    expense.textContent =
        formatCurrency(
            totalExpense
        );


    balance.textContent =
        formatCurrency(
            totalIncome -
            totalExpense
        );

}


/* =====================================================
   EXPENSE GRAPH
===================================================== */

function createGraph() {

    graph.innerHTML = "";


    const days = [];


    for (
        let i = 6;
        i >= 0;
        i--
    ) {

        const date =
            new Date();

        date.setHours(
            0,
            0,
            0,
            0
        );

        date.setDate(
            date.getDate() - i
        );

        days.push(date);

    }


    const values =
        days.map(
            day => {

                const year =
                    day.getFullYear();

                const month =
                    String(
                        day.getMonth() + 1
                    ).padStart(2, "0");

                const date =
                    String(
                        day.getDate()
                    ).padStart(2, "0");

                const dayString =
                    `${year}-${month}-${date}`;


                return transactions
                    .filter(
                        item => {

                            return (
                                item.type ===
                                "expense" &&
                                item.date ===
                                dayString
                            );

                        }
                    )
                    .reduce(
                        (
                            sum,
                            item
                        ) =>
                            sum +
                            Number(item.amount),
                        0
                    );

            }
        );


    const max =
        Math.max(
            ...values,
            1
        );


    days.forEach(
        (day, index) => {

            const container =
                document.createElement(
                    "div"
                );

            container.className =
                "bar-container";


            const value =
                values[index];


            const valueText =
                document.createElement(
                    "div"
                );

            valueText.className =
                "bar-value";

            valueText.textContent =
                value > 0
                    ? formatCurrency(
                        value
                    )
                    : "";


            const bar =
                document.createElement(
                    "div"
                );

            bar.className =
                "bar";


            if (value === 0) {

                bar.classList.add(
                    "zero"
                );

                bar.style.height =
                    "3px";

            } else {

                const height =
                    (
                        value /
                        max
                    ) * 190;

                bar.style.height =
                    Math.max(
                        height,
                        5
                    ) + "px";

            }


            const label =
                document.createElement(
                    "div"
                );

            label.className =
                "bar-label";

            label.textContent =
                day.toLocaleDateString(
                    "en-IN",
                    {
                        weekday: "short"
                    }
                );


            container.appendChild(
                valueText
            );

            container.appendChild(
                bar
            );

            container.appendChild(
                label
            );


            graph.appendChild(
                container
            );

        }
    );

}
